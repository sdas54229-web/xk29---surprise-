const SITE_PASSWORD = "Sayani15";

function checkLockPassword(){

    const input = document.getElementById("lockInput");
    const error = document.getElementById("lockError");

    if(input.value === SITE_PASSWORD){

        document.getElementById("lockScreen").style.display = "none";
        document.getElementById("siteContent").classList.add("unlocked");

        showPage(0);
        playMusic();

    } else {

        error.textContent = "That's not it... try again 💗";
        input.value = "";
        input.focus();

    }

}


let musicStarted = false;

function playMusic(){

    const audio = document.getElementById("bgMusic");
    const btn = document.getElementById("musicButton");

    if(!musicStarted){

        audio.play()
        .then(function(){
            musicStarted = true;
            btn.innerHTML = "🎵 Pause Perfect";
        })
        .catch(function(err){
            console.log("Playback blocked:", err);
            btn.innerHTML = "🎵 Tap again to play";
        });

    } else {

        audio.pause();
        musicStarted = false;
        btn.innerHTML = "🎵 Play Perfect";

    }

}


function answerQuiz(button, isCorrect){

    const card = button.closest(".quiz-card");

    if(card.classList.contains("answered")) return;

    card.classList.add("answered");

    card
    .querySelectorAll(".quiz-option")
    .forEach(function(opt){
        opt.classList.add("incorrect");
    });

    button.classList.remove("incorrect");
    button.classList.add(isCorrect ? "correct" : "incorrect");

    if(!isCorrect){
        card
        .querySelectorAll(".quiz-option")
        .forEach(function(opt){
            if(opt.getAttribute("onclick").includes(",true)")){
                opt.classList.remove("incorrect");
                opt.classList.add("correct");
            }
        });
    }

    const verdict = card.querySelector(".quiz-verdict");
    const explain = card.getAttribute(isCorrect ? "data-right" : "data-wrong") || "";

    verdict.classList.add(isCorrect ? "right" : "wrong");

    verdict.innerHTML =
        (isCorrect ? "✅ Correct! " : "❌ Not quite — ") +
        explain;

}


const pageOrder = [
    { id:"page-hero",   display:"flex"  },
    { id:"personal",    display:"block" },
    { id:"page-story",  display:"block" },
    { id:"page-quiz",   display:"block" },
    { id:"gallery",     display:"block" },
    { id:"page-love",   display:"block" },
    { id:"page-letter", display:"block" },
    { id:"page-final",  display:"flex"  }
];

let currentPageIndex = 0;

function showPage(index){

    if(index < 0) index = 0;
    if(index >= pageOrder.length) index = pageOrder.length - 1;

    currentPageIndex = index;

    pageOrder.forEach(function(page, i){

        const el = document.getElementById(page.id);
        if(!el) return;

        el.style.display = (i === index) ? page.display : "none";

    });

    window.scrollTo({ top: 0, behavior: "instant" });

}

function goToNextPage(){

    showPage(currentPageIndex + 1);
    createHeart();

}

function goToStart(){

    showPage(0);

}

}

function createHeart(){

    const heart=document.createElement("div");

    heart.className="heart";

    const symbols=[
        "❤️",
        "💗",
        "💕",
        "✨",
        "🧿",
        "♡"
    ];

    heart.innerHTML=
    symbols[
        Math.floor(Math.random()*symbols.length)
    ];

    heart.style.left=
    Math.random()*100+"vw";

    heart.style.fontSize=
    (15+Math.random()*18)+"px";

    heart.style.animationDuration=
    (5+Math.random()*5)+"s";

    document.body.appendChild(heart);

    setTimeout(function(){
        heart.remove();
    },10000);

}
